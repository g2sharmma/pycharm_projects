"while updating row with certain conditions, we cannot combine WHERE clause with INSERT"  (to achieve this we have to use UPDATE clause)

INSERT INTO employees (salary)
VALUES (98000.00)
WHERE first_name = "Virat";



UPDATE employees
SET salary = 98000.00
WHERE first_name = "Virat";

5. rank() vs dense_rank() vs row_number()


select first_name, age,
rank() over (order by age) as rank
from customers;


output: ((leaving a gap))

first_name	age	rank
Robert	22	1
David	22	1
John	25	3
Betty	28	4
John	31	5


select first_name, age,
dense_rank() over (order by age desc) as dense_rank
from customers;


output: ((leaving no gap)

first_name	age	dense_rank
John	31	1
Betty	28	2
John	25	3
Robert	22	4
David	22	4


select first_name, age,
row_number() over (order by age desc) as row_number
from customers;

output:

first_name	age	row_number
John	31	1
Betty	28	2
John	25	3
Robert	22	4
David	22	5

  select salesperson, salesamount,
rank() over (partition by salesperson order by salesamount) as rank
from sales
)

select salesperson, salesamount
from salesrecord
where rank > 1;
