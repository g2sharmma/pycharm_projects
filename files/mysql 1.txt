SUBQUERIES:
==========

a query within another query.

SELECT AVG(hourly_pay) FROM employees      (this entire thing becomes on query)

Now if we pass this query to another query then this will be called subquery.



SELCET first_name, last_name, hourly_pay,
		(SELECT AVG(hourly_pay) FROM employees) AS avg_pay
FROM employees;



SELCET first_name, last_name, hourly_pay FROM employees
WHERE hourly_pay > (SELECT AVG(hourly_pay) FROM employees);

stored procedures:
=================

stored procedure is a prepared sql code that you can save.
great use when there's a query you write often.




CREATE PROCEDURE get_customers()
BEGIN
	SELECT * FROM customers;                        (Here the ; gets confused b/w this line and END; line to avoid this we can use delimeter (\\ or $$))
END;


CALL get_customers();




DELIMETER $$
CREATE PROCEDURE get_customers()
BEGIN
	SELECT * FROM customers;                        \
END $$


CALL get_customers();




"passing argument to the procedure"

DELIMITER $$
CREATE PROCEDURE find_customer(IN id INT)
BEGIN
	SELECT * FROM customers
	WHERE customer_id = id;
END $$
DELIMITER ;


CALL find_customer(1);


DROP PROCEDURE find_customer();





"passing multiple arguments to the procedure"


DELIMETER $$
CREATE PROCEDURE find_customer(IN f_name VARCHAR(20), IN l_name VARCHAR(20))
BEGIN
	SELECT * FROM customers
	WHERE first_name = f_name AND last_name = l_name;
END $$
DELIMETER ;


CALL find_customer("Larry", "Lobster");

trigger:
=======


when an event happens to something
for example (INSERT, UPDATE, DELETE)

checks data, handles error, auditing tables.




ALTER TABLE employees
ADD COLUMN salary DECIMAL(10,2) AFTER hourly_pay;


UPDATE employees
SET salary = hourly_pay * 2080
SELECT * FROM employees;                          (here the salary will be updated based on hourly_pay but to apply these changes permanantly we will use trigger)




"lets see how we can use trigger"


CREATE TRIGGER before_hourly_pay_update
BEFORE UPDATE ON employees
FOR EACH ROW
SET NEW.salary = (NEW.hourly_pay * 2080);



"now lets see how trigger works"


UPDATE employees
SET hourly_pay = 50
WHERE employee_id = 1;
SELCET * FROM employees;                   (this will update the employee 1 salary based on new hourly_pay without affecting the others employee)




"lets increment every employee hourly_pay by 1 and update the salary"


UPDATE employees
SET hourly_pay = hourly_pay + 1;
SELECT * FROM employees;



"create trigger for insert"


CREATE TRIGGER before_hourly_pay_insert
BEFORE INSERT ON employees
FOR EACH ROW
SET NEW.salary = (NEW.hourly_pay * 2080);


INSERT INTO employees
VALUES (7, "Jimmy", "Anderson", 100, NULL, "cricketer", "2023-01-07", 5);               (here NULL is set for salary initially, however it will get updated based on hourly_pay)
